# Waffenhaltung von außen prüfen: Rüstungsständer in derselben Armhaltung wie der Spieler (Arm 18° nach vorn),
# von der Seite und von vorn; dazu der Spieler nach ein paar Schritten (Körper zeigt dann in Blickrichtung)
server /gamemode spectator
server /time set 6000
server /weather clear
warte 30
taste F1
server /villa ort 55.5 26 76.5 180 0
server /summon minecraft:armor_stand ~ ~ ~ {ShowArms:1b,NoBasePlate:1b,Rotation:[180f,0f],Pose:{RightArm:[-18f,0f,0f],LeftArm:[-18f,0f,0f]}}
server /item replace entity @e[type=minecraft:armor_stand,sort=nearest,limit=1] weapon.mainhand with stinkendreich:geldkanone
server /item replace entity @e[type=minecraft:armor_stand,sort=nearest,limit=1] weapon.offhand with stinkendreich:goldene_pistole
server /villa ort 58.0 26.2 76.5 90 5
warte 3
bild s01_staender_seite_rechts
server /villa ort 53.0 26.2 76.5 -90 5
warte 2
bild s02_staender_seite_links
server /villa ort 55.5 26.4 73.8 0 8
warte 2
bild s03_staender_vorn
server /villa ort 57.5 27.5 78.5 135 25
warte 2
bild s04_staender_schraeg_hinten
server /kill @e[type=minecraft:armor_stand,sort=nearest,limit=1]
