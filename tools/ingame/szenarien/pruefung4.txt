# 0.2.1 Endkontrolle: Lava-Graben von der Dachterrasse, Privattür, Waffen in der Hand, Rutsche
server /gamemode survival
server /time set 6000
server /weather clear
server /gamerule doDaylightCycle false
warte 30
taste F1
# Lava-Graben: auf der Dachterrasse (gehört zum Haus), Blick nach Süden über Tor und Graben
server /villa ort 55.5 38 72.5 0 28
warte 6
server /execute if block ~ ~-15 ~31 minecraft:lava run say GRABEN_LAVA_OK
bild z01_graben_lava
server /villa ort 40.5 38 72.5 30 30
warte 2
bild z02_graben_lava_seite
# Privattür: als Bewohner öffnen

warte 1
